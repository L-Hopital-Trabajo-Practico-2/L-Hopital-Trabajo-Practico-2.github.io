(?:NO ESPECIFICADO )?DEBIDO"), "TEA / TGD": r"AUTIS|ASPERGER|GENERALIZADOS? DEL DESARROLLO|\bTEA\b|\bTGD\b", "Sensorial (audición / visión)": r"HIPOACUSIA|SORDERA|AUDI|CEGUERA|\bVISION|MACULA|OPTICA|COCLEAR|OJOS", "Síndrome de Down": r"DOWN", "Clínico (respiratorio, cardíaco, otros)": (r"CARDI|RESPIRA|DISNEA|APNEA|PULMON|ASMA|RENAL|LUPUS|DIABET|" r"HIPOTIROID|FALLOT|ILEOSTOMIA|CIRROSIS|HEPAT|PANCREAS|INTESTINO"), } CATS = list(CATEGORIAS) for cat, patron in CATEGORIAS.items(): df[cat] = df["dg_limpio"].str.contains(patron, regex=True) df["n_categorias"] = df[CATS].sum(axis=1) # Agrupamos obras sociales: las dos grandes y el resto df["os_grupo"] = np.where(df["Obra Social"].isin(["OSECAC", "INCLUIR"]), df["Obra Social"], "Resto") # --------------------------------------------------------------- # Base limpia # --------------------------------------------------------------- def titulo(t): return " ".join(p.capitalize() if p not in ("DE", "DEL", "LA", "LOS") or i == 0 else p.lower() for i, p in enumerate(t.split())) def localidad(fila): if fila["zona"] == "CABA sin barrio": return "Capital Federal" if fila["zona"] == "CABA con barrio": return f"Capital Federal, {titulo(fila['barrio'])}" return f"Provincia de Bs.
